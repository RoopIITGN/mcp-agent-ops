# Pinned to what the code actually imports today. This grows one phase at a
# time rather than up front: a requirements file listing packages the repo does
# not use yet is a list of things a reader has to check against the code.
#

# tests/test_tools.py have no dependencies beyond pydantic, which is what lets
# most of the CI gate in Phase 6 run without credentials or a model.

mcp>=2.2,<3
